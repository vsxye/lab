import numpy as np
import matplotlib.pyplot as plt

def f(x):
    return 20 + sum(x*x - 10*np.cos(2*np.pi*x))

n = 20
pop = np.random.uniform(-5.12, 5.12, (n, 2))
history = []

for t in range(200):
    fitness = np.array([f(x) for x in pop])
    order = np.argsort(fitness)
    pop = pop[order]
    best = pop[0].copy()

    rates = np.linspace(1, 0, n)

    for i in range(1, n):
        for j in range(2):
            if np.random.rand() < rates[i]:
                k = np.random.choice(n, p=rates / rates.sum())
                pop[i][j] = pop[k][j]

            if np.random.rand() < 0.05:
                pop[i][j] = np.random.uniform(-5.12, 5.12)

    pop = np.clip(pop, -5.12, 5.12)
    pop[-1] = best
    history.append(f(best))

best = min(pop, key=f)

print("Best position:", best)
print("Minimum value:", round(f(best), 4))

plt.plot(history)
plt.xlabel("Iteration")
plt.ylabel("Best Fitness")
plt.title("BBO Convergence")
plt.grid()
plt.show()
