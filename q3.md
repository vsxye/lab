import random
import math

def f(x):
    return x * x

n = 10
pos = [random.uniform(-10, 10) for i in range(n)]
vel = [random.uniform(-1, 1) for i in range(n)]
pbest = pos[:]
gbest = min(pos, key=f)

for i in range(100):
    for j in range(n):
        vel[j] = 0.5 * vel[j] + 1.5 * random.random() * (pbest[j] - pos[j]) + 1.5 * random.random() * (gbest - pos[j])
        pos[j] += vel[j]

        if f(pos[j]) < f(pbest[j]):
            pbest[j] = pos[j]

        if f(pbest[j]) < f(gbest):
            gbest = pbest[j]

print("Best position:", round(gbest, 4))
print("Minimum value:", round(f(gbest), 4))
