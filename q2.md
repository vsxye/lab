import random
import math
import matplotlib.pyplot as plt

cities = [(random.randint(0, 100), random.randint(0, 100)) for i in range(5)]
n = len(cities)

def distance(a, b):
    return math.dist(cities[a], cities[b])

def cost(route):
    return sum(distance(route[i], route[(i+1) % n]) for i in range(n))

route = list(range(n))
random.shuffle(route)
best = route[:]
T = 1000
history = []

for i in range(1000):
    a, b = sorted(random.sample(range(n), 2))
    new = route[:]
    new[a:b+1] = new[a:b+1][::-1]

    old_cost = cost(route)
    new_cost = cost(new)

    if new_cost < old_cost or random.random() < math.exp((old_cost-new_cost)/T):
        route = new

    if cost(route) < cost(best):
        best = route[:]

    history.append(cost(best))
    T *= 0.99

print("Cities:", cities)
print("Best route:", best)
print("Minimum distance:", round(cost(best), 2))

plt.plot(history)
plt.xlabel("Iteration")
plt.ylabel("Best Distance")
plt.title("TSP Simulated Annealing")
plt.show()

x = [cities[i][0] for i in best] + [cities[best[0]][0]]
y = [cities[i][1] for i in best] + [cities[best[0]][1]]

plt.plot(x, y, marker="o")
plt.title("Best TSP Route")
plt.show()
