
import random
import math

cities = [(random.randint(0, 100), random.randint(0, 100)) for i in range(5)]

def distance(a, b):
    return math.dist(cities[a], cities[b])

def cost(route):
    return sum(distance(route[i], route[(i+1) % len(route)]) for i in range(len(route)))

route = list(range(len(cities)))
random.shuffle(route)
best = route[:]
T = 1000

for i in range(1000):
    a, b = sorted(random.sample(range(len(route)), 2))
    new = route[:]
    new[a:b+1] = new[a:b+1][::-1]

    if cost(new) < cost(route) or random.random() < math.exp((cost(route)-cost(new))/T):
        route = new

    if cost(route) < cost(best):
        best = route[:]

    T *= 0.99

print("Cities:", cities)
print("Best route:", best)
print("Minimum distance:", round(cost(best), 2))
