
import numpy as np
import matplotlib.pyplot as plt

def f(x):
    return 20 + sum(x*x - 10*np.cos(2*np.pi*x))

n = 30
pop = np.random.uniform(-5.12, 5.12, (n, 2))
alpha = 0.2
beta0 = 1.0
gamma = 1.0
history = []

best = pop[np.argmin([f(x) for x in pop])].copy()

for t in range(200):
    intensity = np.array([f(x) for x in pop])

    for i in range(n):
        for j in range(n):
            if intensity[j] < intensity[i]:
                r = np.linalg.norm(pop[i] - pop[j])
                beta = beta0 * np.exp(-gamma * r**2)
                pop[i] += beta * (pop[j] - pop[i]) + alpha * (np.random.rand(2) - 0.5)
                pop[i] = np.clip(pop[i], -5.12, 5.12)

    alpha *= 0.99
    best_now = pop[np.argmin([f(x) for x in pop])]

    if f(best_now) < f(best):
        best = best_now.copy()

    history.append(f(best))

print("Best position:", best)
print("Minimum value:", round(f(best), 4))

plt.plot(history)
plt.xlabel("Iteration")
plt.ylabel("Best Fitness")
plt.title("Firefly Algorithm Convergence")
plt.grid()
plt.show()
